# 2. TRANSIENT RAM STORAGE (AUTO-DELETES IN 30 MINUTES)
# Multi-user storage dictionary kept entirely in server memory. No logs are saved.
PRIVATE_MEMORY_STORE = {}
SESSION_TIMEOUT_SECONDS = 1800 # Exactly 30 minutes of inactivity before deletion

AI_CORE_PERSONA = """
You are Sachii AI, an elite, publicly accessible information verification agent. You provide a safe, completely unmonitored, deeply private, and secure public space for users worldwide to track down truths.
You DO NOT sell, store, log, or share user data. All operations happen in transient RAM and vanish forever after 30 minutes of inactivity.

2. NO BIAS & NO ARGUMENTS: You must remain completely unbiased. Do not judge, take political/ideological sides, argue, or engage in debates. Provide an inclusive, universal safe space for all members of the public.
3. DE-ESCALATION: If a user displays anger, distress, frustration, or hostility, actively use gentle, calm, and neutral language to de-escalate the tension and work towards a peaceful resolution.
4. PUBLIC CRISIS SUPPORT: If a user indicates they are hurting, in danger, or experiencing a mental health crisis, immediately and prominently provide verified, real-world helpline resources (such as 988 Crisis Lifelines, text lines, etc.) in a clear, supportive format.

You must look up information and actively cross-examine it for contradictions, bias, or gaps. If the information cannot be fully cross-referenced, lacks concrete evidence, or contains conflicting accounts, you MUST explicitly state to the user:
"VERIFICATION STATUS: UNVERIFIABLE"
Follow this phrase with a thorough explanation of what parts of the claim could not be independently cross-examined and confirmed.

def purge_expired_sessions():
    """Loops through RAM memory and explicitly wipes out sessions inactive for 30+ minutes"""
    current_time = time.time()
    expired_sessions = [
        sid for sid, data in PRIVATE_MEMORY_STORE.items()
        if (current_time - data["last_activity"]) > SESSION_TIMEOUT_SECONDS
    ]
    for sid in expired_sessions:
        # Permanently drops the object from RAM memory allocation
        del PRIVATE_MEMORY_STORE[sid]

def get_or_update_chat_session(session_id):
    # Wipe old data before serving new requests to manage public server memory
    purge_expired_sessions()
    current_time = time.time()
   
    if session_id not in PRIVATE_MEMORY_STORE:
        chat_instance = client.chats.create(
            model="gemini-2.5-flash",
            config=types.GenerateContentConfig(
                system_instruction=AI_CORE_PERSONA,
                tools=[types.Tool(google_search=types.GoogleSearch())],
                temperature=0.2
            )
        )
        PRIVATE_MEMORY_STORE[session_id] = {
            "chat": chat_instance,
            "last_activity": current_time
        }
    else:
        # Reset the 30-minute countdown clock on fresh activity
        PRIVATE_MEMORY_STORE[session_id]["last_activity"] = current_time
       
    return PRIVATE_MEMORY_STORE[session_id]["chat"]

# 4. ROUTING GATEWAYS
@app.route('/')
def main_interface():
    return render_template_string(UI_TEMPLATE)

@app.route('/chat-gateway', methods=['POST'])
def chat_gateway():
    session_id = request.form.get('session_id', 'anonymous_identity')
    user_message = request.form.get('message', '')
    uploaded_image = request.files.get('image')
   
    active_chat = get_or_update_chat_session(session_id)
    payload_contents = []
   
    if uploaded_image:
        raw_bytes = uploaded_image.read()
        image_payload_part = types.Part.from_bytes(
            data=raw_bytes,
            mime_type=uploaded_image.content_type
        )
        payload_contents.append(image_payload_part)
       
    if user_message:
        payload_contents.append(user_message)
       
    if not payload_contents:
        return jsonify({"response": "I did not receive any input to process."})
       
    try:
        api_response = active_chat.send_message(payload_contents)
        return jsonify({"response": api_response.text})
    except Exception as runtime_exception:
        return jsonify({"response": f"An unexpected system exception occurred: {str(runtime_exception)}"})
